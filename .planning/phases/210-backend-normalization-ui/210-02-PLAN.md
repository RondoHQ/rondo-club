**1. Add `formatPhoneForDisplay()` to `src/utils/formatters.js`:**

Create a new exported function that converts E.164 phone numbers to readable Dutch format:

```javascript
export function formatPhoneForDisplay(phone) {
  if (!phone) return '';
  const cleaned = String(phone).trim();
  if (!cleaned) return '';
  // Dutch mobile: +316xxxxxxxx -> 06-xxxxxxxx
  if (/^\+316\d{8}$/.test(cleaned)) {
    return `06-${cleaned.slice(4)}`;
  }
  // Dutch landline: +31 followed by area code + subscriber
  // 3-digit area codes: 010, 020, 030, 040, 050, 070, 071-079, 088
  // All other are 4-digit area codes (e.g., 0251, 0318)
  if (/^\+31\d{9}$/.test(cleaned)) {
    const national = cleaned.slice(3); // 9 digits after +31
    const threeDigitAreas = ['10','13','14','15','20','23','24','26','30','33','35','36','38','40','43','45','46','50','53','55','58','70','71','72','73','74','75','76','77','78','79','88'];
    const areaPrefix = national.slice(0, 2);
    if (threeDigitAreas.includes(areaPrefix)) {
      // 3-digit area code (0xx): 0xx-xxxxxxx
      return `0${national.slice(0, 2)}-${national.slice(2)}`;
    }
    // 4-digit area code (0xxx): 0xxx-xxxxxx
    return `0${national.slice(0, 3)}-${national.slice(3)}`;
  }
  // Non-NL international: show with spaces after country code
  if (/^\+\d+$/.test(cleaned) && cleaned.length > 4) {
    // Find country code (1-3 digits after +), then add spaces every 3 digits
    // Simple approach: +XX XXXXXXXXX for known 2-digit codes, +X for 1-digit, +XXX for 3-digit
    // For simplicity, just put a space after the first 3 chars (e.g., +49 123456789)
    return `${cleaned.slice(0, 3)} ${cleaned.slice(3)}`;
  }
  // Fallback: return as-is
  return cleaned;
}
```

Note: The `threeDigitAreas` list covers all Dutch 3-digit area codes. Everything else under +31 with 9 subscriber digits is a 4-digit area code.
